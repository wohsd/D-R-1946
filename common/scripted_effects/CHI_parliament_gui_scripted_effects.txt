# The following scripted effect does the main job of generating the gui, by adding the seats for each party in a grid.
# This should be called every time you need to update the gui, when you add/subtract seats for eCHI_parliament_x_arrayample.
# Read the comments if you're interested in how it works
reload_CHI_parliament_gui = {
	# Clears what's currently on the gui, if there's anything there.
	clear_array = CHI_parliament_main_array

	# The array CHI_temp_seat is duplicated from the array CHI_parliament_seat_array.
	for_each_loop = {
		array = CHI_parliament_seat_array

		add_to_array = { CHI_temp_seat = CHI_parliament_seat_array^i }
	}

	# It's a suprise tool for later
	set_temp_variable = { CHI_temp_index_parliament_x_array_1 = 0 }
	set_temp_variable = { CHI_temp_index_parliament_x_array_2 = 1 }

	# Set CHI_temp_size as the highest indeCHI_parliament_x_array in the array, highest indeCHI_parliament_x_array is the highest value CHI_parliament_x_array in CHI_temp_array^CHI_parliament_x_array.
	set_temp_variable = { CHI_temp_size = CHI_temp_seat^num }
	subtract_from_temp_variable = { CHI_temp_size = 1 }

	# What this does is loop through the CHI_temp_seat array, adding the current value to the neCHI_parliament_x_arrayt value on the list.
	# CHI_temp_seat^0 is the first value on the list, CHI_temp_seat^1 is the neCHI_parliament_x_arrayt and so forth.
	# the loop ends when CHI_temp_index_parliament_x_array_1 is equal to the size of the array - 1 (basically the maCHI_parliament_x_array indeCHI_parliament_x_array).
	while_loop_effect = {
		limit = {
			check_variable = { CHI_temp_index_parliament_x_array_1 < CHI_temp_size }
		}
		# The suprise tool starts out as CHI_temp_seat^0 and CHI_temp_seat^1 and then is incremented every loop , adding the current value to the neCHI_parliament_x_arrayt value on the list.
		add_to_variable = { CHI_temp_seat^CHI_temp_index_parliament_x_array_2 = CHI_temp_seat^CHI_temp_index_parliament_x_array_1 }
		add_to_temp_variable = { CHI_temp_index_parliament_x_array_1 = 1 }
		add_to_temp_variable = { CHI_temp_index_parliament_x_array_2 = 1 }
	}

	# resize_array fills the CHI_parliament_member_array with the id of the CHI_parliament member CHI_parliament_x_array amount of times where CHI_parliament_x_array is the number of seats the member has.
	# so the array will look like this basically:
	# 1,1,1,1,1,1,1,...(repeated 90 times),2,2,2,2,2,2,2(repeated 35 times),3,3,3.. etc.
	for_loop_effect = {
		start = 0
		end = CHI_temp_seat^num

		resize_array = { array = CHI_parliament_main_array value = CHI_parliament_member_array^v size = CHI_temp_seat^v }
	}

	# Clear the CHI_temp_seat array used previously since it's not needed anymore.
	clear_array = CHI_temp_seat
}

# Bonus code for adding/subtracting seats
modify_CHI_parliament_seat = {
	subtract_from_temp_variable = { CHI_temp_member = 1 }

	# check to see if result overflows
	set_temp_variable = { CHI_temp = CHI_parliament_seat_array^CHI_temp_member }
	add_to_temp_variable = { CHI_temp = CHI_temp_seats }

	if = {
		limit = {
			check_variable = { CHI_temp > CHI_parliament_main_array^num }
		}
		subtract_from_temp_variable = { CHI_temp = CHI_parliament_main_array^num }
		subtract_from_temp_variable = { CHI_temp_seats = CHI_temp }
	}
	else_if = {
		limit = {
			check_variable = { CHI_temp < 0 }
		}
		multiply_temp_variable = { CHI_temp = -1 }
		subtract_from_temp_variable = { CHI_temp_seats = CHI_temp }
	}

	# adds all parties to CHI_temp_array
	for_loop_effect = {
		start = 0
		end = CHI_parliament_seat_array^num

		add_to_temp_array = { CHI_temp_array = v }
	}

	# remove from list of parties the party to be added/subtracted
	remove_from_temp_array = { array = CHI_temp_array value = CHI_temp_member }

	# give party new seat value
	add_to_variable = { CHI_parliament_seat_array^CHI_temp_member = CHI_temp_seats }

	# if its add
	if = {
		limit = {
			check_variable = { CHI_temp_seats > 0 }
		}
		# loop until seats to be added is 0
		while_loop_effect = {
			limit = {
				check_variable = { CHI_temp_seats > 0 }
			}

			# loops through remaining parties, subtracting 1 seat each time
			for_loop_effect = {
				start = 0
				end = CHI_temp_array^num

				if = {
					limit = {
						check_variable = { CHI_parliament_seat_array^CHI_temp_array^v > 0 } # party must have more than 0 seats to be subtracted
						check_variable = { CHI_temp_seats > 0 } # seats to be added is bigger than 0 currently
					}
					subtract_from_variable = { CHI_parliament_seat_array^CHI_temp_array^v = 1 } # subtract from party 1 seat
					subtract_from_temp_variable = { CHI_temp_seats = 1 } # subtract from seats to be added
				}
			}
		}
	}
	# if its subtract
	else_if = {
		limit = {
			check_variable = { CHI_temp_seats < 0 }
		}
		# loop until seats to be subtracted is 0
		while_loop_effect = {
			limit = {
				check_variable = { CHI_temp_seats < 0 }
			}

			# loops through remaining parties, adding 1 seat each time
			for_loop_effect = {
				start = 0
				end = CHI_temp_array^num

				if = {
					limit = {
						check_variable = { CHI_parliament_seat_array^CHI_temp_array^v < CHI_parliament_main_array^num } # party must have less than maCHI_parliament_x_array seats to be added
						check_variable = { CHI_temp_seats < 0 } # seats to be subtracted is less than 0 currently
					}
					add_to_variable = { CHI_parliament_seat_array^CHI_temp_array^v = 1 } # add to party 1 seat
					add_to_temp_variable = { CHI_temp_seats = 1 } # add to seats to be subtracted
				}
			}
		}
	}
	# do nothing if 0

	# Modify what shows up in the gui
	reload_CHI_parliament_gui = yes
}
