	name = CHIGetParliamentPartyName
	text = {
		trigger = {
			set_temp_variable = { CHI_temp_seat = CHI_parliament_main_index }
			add_to_temp_variable = { CHI_temp_seat = 1 }
			check_variable = { CHI_temp_seat > 0 }
		}
		localization_key = CHI_parliament_seat_text
	}
