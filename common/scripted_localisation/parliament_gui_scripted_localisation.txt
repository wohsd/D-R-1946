defined_text = {
	name = JAPGetGovernmentStatus
	
	# 1. Priority: Coalition
	text = { 
		trigger = { has_country_flag = coalition_active } 
		localization_key = "government_coalition" 
	}
	
	# 2. 268 seats and above (2/3 Supermajority)
	text = { 
		trigger = { check_variable = { var = temp_ruling_party_seats value = 125 compare = greater_than } } 
		localization_key = "government_supermajority" 
	}
	
	# 3. Between 201 and 267 (Absolute Majority)
	text = { 
		trigger = { check_variable = { var = temp_ruling_party_seats value = 90 compare = greater_than } } 
		localization_key = "government_absolute_majority" 
	}
	
	# 4. 100 and below (Minority Government)
	text = { 
		trigger = { always = yes } 
		localization_key = "government_minority" 
	}
}

	}

		trigger = { always = yes }

}

# Since index starts as 0, i can get the current seat number by jut adding it by one and then displaying it.
defined_text = {
	name = GetParliamentSeatName
	text = {
		trigger = {
			set_temp_variable = { temp_seat = parliament_main_index }
			add_to_temp_variable = { temp_seat = 1 }
			check_variable = { temp_seat > 0 }
		}
		localization_key = parliament_seat_text
	}
}
