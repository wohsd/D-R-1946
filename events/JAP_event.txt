# 越南危机事件
# ====================================
add_namespace = JAP.vietnam_crisis

# 事件1：加急生产完成
country_event = {
	id = JAP.vietnam_crisis.1
	title = JAP.vietnam_crisis.1.t
	desc = JAP.vietnam_crisis.1.d
	picture = GFX_event_japan_pacific_war
	is_triggered_only = yes
	fire_only_once = no
	
	immediate = {
		clr_country_flag = urgent_bomber_production_active
		add_consumer_goods_factor = -0.10
		add_variable = { bombers_available = 5 }
		if = {
			limit = {
				check_variable = { which = bombers_available value > 100 }
			}
			set_variable = { bombers_available = 100 }
		}
	}
	
	option = {
		name = JAP.vietnam_crisis.1.a
	}
}

# 事件2：越南危机开始

	id = JAP.vietnam_crisis.2

	is_triggered_only = yes
	fire_only_once = yes
	
	immediate = {
		set_country_flag = vietnam_crisis_active
		set_variable = { bombers_available = 100 }
	}
	
	option = {
		name = JAP.vietnam_crisis.2.a
	}

# 事件3：大藏省的紧急报告
country_event = {
	id = JAP.vietnam_crisis.3
	title = JAP.vietnam_crisis.3.t
	desc = JAP.vietnam_crisis.3.d
	picture = GFX_event_japan_pacific_war
	fire_only_once = yes
	
	trigger = {
		tag = JAP
		has_country_flag = vietnam_crisis_active
	}
	
	mean_time_to_happen = {
		days = 120
		modifier = {
			factor = 0.5
			has_country_flag = vietnam_crisis_active
		}
	}
	
	option = {
		name = JAP.vietnam_crisis.3.a
		add_political_power = -25
		add_consumer_goods_factor = -0.10
		add_military_factory = 2
		set_country_flag = finance_emergency_a
	}
}
